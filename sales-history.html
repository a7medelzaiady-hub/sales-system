// ===== تصدير الإكسيل - مع التعديلات المطلوبة ===== document.getElementById('exportExcelBtn').onclick = () => { if (!allInvoices.length) { window.showCustomUi({ title: "تنبيه", message: "لا توجد بيانات لتصديرها بعد.", icon: "info" }); return; } let excelRows = []; allInvoices.forEach(inv => { const invDate = parseInvoiceDate(inv); const customerName = inv.customerName || 'النقدى'; // 1. أصناف الفاتورة - فصل الجركن والكرتونة if (inv.items && inv.items.length > 0) { inv.items.forEach(itm => { let qtyCarton = 0; let qtyJarkan = 0; let priceCarton = 0; let priceJarkan = 0; const itemName = itm.name || ""; const rawQty = Number(itm.qty) || 0; const rawPrice = Number(itm.price) || 0; if (itemName.includes("جركن") || itemName.includes("جركان")) { // صنف جركن يُسجل في أعمدة الجركن فقط qtyJarkan = rawQty; priceJarkan = rawPrice; } else { // صنف كرتونة يُسجل في أعمدة الكرتونة فقط qtyCarton = rawQty; priceCarton = rawPrice; } excelRows.push({ "التاريخ": invDate, "الحركة": "بيع", "اسم العميل": customerName, "الصنف": itemName, "عدد الجركن": qtyJarkan, "عدد الكرتونة": qtyCarton, "سعر الجركن": priceJarkan, "سعر الكرتونة": priceCarton }); }); } // 2. السداد النقدي كسطر مستقل if ((inv.calculatedCash || 0) > 0) { excelRows.push({ "التاريخ": invDate, "الحركة": "سداد منه نقدى", "اسم العميل": customerName, "الصنف": "سداد نقدى", "عدد الجركن": 0, "عدد الكرتونة": 1, "سعر الجركن": 0, "سعر الكرتونة": inv.calculatedCash }); } // 3. سداد المحفظة كسطر مستقل if ((inv.calculatedWallet || 0) > 0) { excelRows.push({ "التاريخ": invDate, "الحركة": "سداد منه محفظه", "اسم العميل": customerName, "الصنف": "سداد محفظه", "عدد الجركن": 0, "عدد الكرتونة": 1, "سعر الجركن": 0, "سعر الكرتونة": inv.calculatedWallet }); } }); const ws = XLSX.utils.json_to_sheet(excelRows); ws['!dir'] = "rtl"; const wb = XLSX.utils.book_new(); XLSX.utils.book_append_sheet(wb, ws, "المبيعات والأرشيف"); XLSX.writeFile(wb, "سجل_المبيعات_التفصيلي.xlsx"); };
